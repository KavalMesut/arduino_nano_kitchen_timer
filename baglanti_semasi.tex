\section{Giris ve Onemli Guvenlik Uyarilari}
Bu belge, Arduino Nano tabanli, TM1637 7 segment gostergeli, EC11 doner enkoder kontrollu, D4184 MOSFET suruculu ve \textbf{18650 2200 mAh Li-ion Pil + LX-LCBST (TP4056 + DC-DC 5V Yukseltici)} sarj sistemli mutfak geri sayim alarminin baglanti detaylarini icermektedir.

\subsection{TRIMPOT KALIBRASYON UYARISI (COK ONEMLI!)}
\textbf{LX-LCBST modulunun VO+ / VO- cikislarini Arduino, ekran veya MOSFET kartina BAGLAMADAN ONCE;}
Type-C kablosunu veya pili takip bir multimetre (avometre) ile \textbf{VO+} cikis voltajini olcun. Modul uzerindeki minyatur trimpotu cevirerek cikis voltajini \textbf{tam 5.0 Volt'a} ayarlayin! Fabrikasyon olarak 9V/12V ayarli gelebilir; ayarlamadan baglarsaniz Arduino kartiniz ve ekraniniz yanabilir!

\subsection{Hoparlor ve MOSFET Emniyet Direnci}
4 Ohm 5W gucundeki hoparlorun yuksek anlik akim cekerek Arduino'ya reset attirmasini ve hoparlor bobininin zarar gormesini engellemek amaciyla hoparlor hattina seri olarak \textbf{10--15 Ohm (en az 1W)} koruma direnci baglanmalidir. 

\textbf{Direnc Birlestirme Ipucu:} Elinizde sadece 1/4W gucunde standart direncler varsa, \textbf{4 adet 47 Ohm 1/4W direnci birbirine paralel lehimleyerek} 11.7 Ohm degerinde 1W gucunde son derece guvenli ve isinmayan bir direnc blogu olusturabilirsiniz.

\section{Adim Adim Baglanti Talimatlari}

\begin{enumerate}
    \item \textbf{Guc Kaynagi ve Sarj Sistemi (LX-LCBST + 18650 Pil):}
    \begin{itemize}
        \item \textbf{B+} $\rightarrow$ 18650 Pilin Arti (+) kutbuna ve ayni zamanda Arduino Nano \textbf{A0} pinine (Batarya voltaji okuma).
        \item \textbf{B-} $\rightarrow$ 18650 Pilin Eksi (-) kutbuna.
        \item \textbf{VO+} $\rightarrow$ Surgulu Guc Anahtarinin (ON/OFF) bacak 1'ine. Anahtarin bacak 2'si ortak \textbf{+5V} rayina baglanir.
        \item \textbf{VO-} $\rightarrow$ Ortak \textbf{GND} sasi rayina.
        \item \textbf{IN+} (Type-C Girisi) $\rightarrow$ 10k koruma direnci uzerinden Arduino Nano \textbf{A1} pinine (Sarj algilama).
    \end{itemize}

    \item \textbf{Ciplak Doner Enkoder (EC11) Baglantilari:}
    \begin{itemize}
        \item \textbf{Out A} (Ust Pin - 3'lu grup) $\rightarrow$ Arduino Nano \textbf{D2}
        \item \textbf{GND} (Orta Pin - 3'lu grup) $\rightarrow$ Ortak \textbf{GND}
        \item \textbf{Out B} (Alt Pin - 3'lu grup) $\rightarrow$ Arduino Nano \textbf{D3}
        \item \textbf{Switch} (Sol Pin - 2'li grup) $\rightarrow$ Arduino Nano \textbf{D4}
        \item \textbf{GND} (Sag Pin - 2'li grup) $\rightarrow$ Ortak \textbf{GND}
    \end{itemize}

    \item \textbf{TM1637 Gosterge Baglantilari:}
    \begin{itemize}
        \item \textbf{CLK} $\rightarrow$ Arduino Nano \textbf{D5}
        \item \textbf{DIO} $\rightarrow$ Arduino Nano \textbf{D6}
        \item \textbf{VCC} $\rightarrow$ Ortak \textbf{+5V}
        \item \textbf{GND} $\rightarrow$ Ortak \textbf{GND}
    \end{itemize}

    \item \textbf{D4184 MOSFET Surucu \& Hoparlor Baglantilari:}
    \begin{itemize}
        \item \textbf{(PWM) +} Sinyal $\rightarrow$ Arduino Nano \textbf{D9}
        \item \textbf{Sinyal GND} $\rightarrow$ Ortak \textbf{GND}
        \item \textbf{DC+ (Klemens Alt Sag)} $\rightarrow$ Ortak \textbf{+5V}
        \item \textbf{DC- (Klemens Alt Sol)} $\rightarrow$ Ortak \textbf{GND}
        \item \textbf{Cikis+ (Klemens Ust Sag)} $\rightarrow$ Hoparlor (+)
        \item \textbf{Cikis- (Klemens Ust Sol)} $\rightarrow$ Paralel Direnc Bloğu ucu 1
        \item Paralel Direnc Bloğu ucu 2 $\rightarrow$ Hoparlor (-)
    \end{itemize}
\end{enumerate}
